
=== TABLE: purchase ===
  id: text NOT NULL
  supplier_id: text
  branch_id: text NOT NULL
  invoice_no: text
  invoice_date: date
  subtotal: numeric NOT NULL
  discount: numeric NOT NULL
  taxable_value: numeric NOT NULL
  cgst: numeric NOT NULL
  sgst: numeric NOT NULL
  igst: numeric NOT NULL
  total: numeric NOT NULL
  payment_method: text
  credit_terms_days: integer
  due_date: date
  status: text NOT NULL
  created_by: text
  approved_by: text
  posted_at: timestamp with time zone
  metadata: jsonb
  created_at: timestamp with time zone NOT NULL
  updated_at: timestamp with time zone NOT NULL
  deleted_at: timestamp with time zone
  Constraints:
    [f] purchase_branch_id_fkey: FOREIGN KEY (branch_id) REFERENCES branch(id) ON DELETE RESTRICT
    [p] purchase_pkey: PRIMARY KEY (id)
    [c] purchase_status_check: CHECK ((status = ANY (ARRAY['DRAFT'::text, 'SUBMITTED'::text, 'APPROVED'::text, 'POSTED'::text])))
    [f] purchase_supplier_id_fkey: FOREIGN KEY (supplier_id) REFERENCES supplier(id) ON DELETE RESTRICT
    [u] uq_purchase_invoice_supplier_branch: UNIQUE NULLS NOT DISTINCT (branch_id, supplier_id, invoice_no)

=== TABLE: purchase_item ===
  id: text NOT NULL
  purchase_id: text NOT NULL
  product_id: text
  variant_id: text
  description: text NOT NULL
  quantity: numeric NOT NULL
  rate: numeric NOT NULL
  discount: numeric NOT NULL
  taxable_value: numeric NOT NULL
  cgst_rate: numeric
  sgst_rate: numeric
  igst_rate: numeric
  cgst: numeric
  sgst: numeric
  igst: numeric
  total: numeric NOT NULL
  hsn_code: text
  created_at: timestamp with time zone NOT NULL
  updated_at: timestamp with time zone NOT NULL
  deleted_at: timestamp with time zone
  Constraints:
    [c] purchase_item_discount_check: CHECK ((discount >= (0)::numeric))
    [p] purchase_item_pkey: PRIMARY KEY (id)
    [f] purchase_item_purchase_id_fkey: FOREIGN KEY (purchase_id) REFERENCES purchase(id) ON DELETE CASCADE
    [c] purchase_item_quantity_check: CHECK ((quantity > (0)::numeric))
    [c] purchase_item_rate_check: CHECK ((rate >= (0)::numeric))

=== TABLE: purchase_payment ===
  id: text NOT NULL
  purchase_id: text NOT NULL
  amount: numeric NOT NULL
  method: text NOT NULL
  payment_date: date NOT NULL
  reference: text
  created_by: text
  created_at: timestamp with time zone NOT NULL
  updated_at: timestamp with time zone NOT NULL
  deleted_at: timestamp with time zone
  Constraints:
    [c] purchase_payment_amount_check: CHECK ((amount > (0)::numeric))
    [c] purchase_payment_method_check: CHECK ((method = ANY (ARRAY['CASH'::text, 'UPI'::text, 'BANK_TRANSFER'::text, 'CHEQUE'::text, 'CARD'::text, 'OTHER'::text])))
    [p] purchase_payment_pkey: PRIMARY KEY (id)
    [f] purchase_payment_purchase_id_fkey: FOREIGN KEY (purchase_id) REFERENCES purchase(id) ON DELETE CASCADE

=== TABLE: supplier ===
  id: text NOT NULL
  name: text NOT NULL
  gstin: text
  phone: text
  email: text
  address: text
  city: text
  state: text
  branch_id: text
  is_active: boolean NOT NULL
  created_by: text
  metadata: jsonb
  created_at: timestamp with time zone NOT NULL
  updated_at: timestamp with time zone NOT NULL
  deleted_at: timestamp with time zone
  Constraints:
    [f] supplier_branch_id_fkey: FOREIGN KEY (branch_id) REFERENCES branch(id) ON DELETE SET NULL
    [p] supplier_pkey: PRIMARY KEY (id)

=== TABLE: branch_inventory ===
  id: text NOT NULL
  product_id: text NOT NULL
  variant_id: text NOT NULL
  quantity: integer NOT NULL
  reserved: integer NOT NULL
  branch_id: text NOT NULL
  created_at: timestamp with time zone NOT NULL
  updated_at: timestamp with time zone NOT NULL
  deleted_at: timestamp with time zone
  Constraints:
    [f] branch_inventory_branch_id_foreign: FOREIGN KEY (branch_id) REFERENCES branch(id) ON UPDATE CASCADE ON DELETE CASCADE
    [p] branch_inventory_pkey: PRIMARY KEY (id)
