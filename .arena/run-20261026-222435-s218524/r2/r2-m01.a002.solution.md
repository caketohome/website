```typescript
// 1. Strict Prefix Validation (CGST Rule 46: max 16 alphanumeric characters)
const rawPrefix = isPartner 
  ? `PRT-${String(partnerId).slice(-4).toUpperCase()}` 
  : (order.invoice_prefix || order.metadata?.invoice_prefix || snap.invoice_prefix || "CTH");
const prefix = String(rawPrefix || "").trim();

if (!prefix || !/^[A-Za-z0-9\-_]{1,16}$/.test(prefix)) {
  return {
    success: false,
    error: `Cannot generate statutory invoice: Invoice series prefix '${prefix}' is invalid under CGST Rule 46. Prefix must be 1 to 16 alphanumeric characters (including hyphens and underscores).`,
    code: "INVOICE_PREFIX_INVALID",
  };
}

// 2. Fail-Closed Statutory Series Conflict & Verification Check
if (!isPartner) {
  try {
    const { dbQuery } = await import("../db");
    const fy = 2026;
    const seqRes = await dbQuery(
      `SELECT prefix, gstin, fiscal_year FROM public.invoice_sequence 
       WHERE prefix = $1 AND fiscal_year = $2`,
      [prefix, fy]
    );
    if (seqRes.rows.length > 0 && seqRes.rows[0].gstin !== sellerGstin) {
      return {
        success: false,
        error: `Cannot generate statutory invoice: Invoice series prefix '${prefix}' in FY ${fy} is legally bound to GSTIN '${seqRes.rows[0].gstin}' under CGST Rule 46. It cannot be used for active seller GSTIN '${sellerGstin}'. Statutory invoice allocation blocked pending CA / Tax Counsel authorized series designation.`,
        code: "INVOICE_SERIES_PREFIX_CONFLICT",
      };
    }
  } catch (seqCheckErr: any) {
    if (seqCheckErr && typeof seqCheckErr === "object" && seqCheckErr.code === "INVOICE_SERIES_PREFIX_CONFLICT") {
      return seqCheckErr;
    }
    const safeMsg = String(seqCheckErr?.message || "Database connection error").replace(/:\/\/.*@/, "://[REDACTED]@");
    return {
      success: false,
      error: `Cannot generate statutory invoice: Database sequence verification failed (${safeMsg}). Statutory invoice allocation blocked fail-closed under CGST Rule 46.`,
      code: "INVOICE_SEQUENCE_VERIFICATION_FAILED",
    };
  }
}
```

## 3. Order & Payment Preservation
When invoicing fails:
- Returns `{ success: false, error: ..., code: ... }`.
- In `order-lifecycle-service.ts`, the failure is logged and does NOT block delivery or roll back order data.
- The order remains intact with status `delivered` / `confirmed` in PostgreSQL `"order"`.
